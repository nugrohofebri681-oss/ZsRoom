# Review/export reproduction

Temporary workspaces and ports in scripts are capture-session paths. These scripts do not change game source.

## Native export
```js
import fs from 'node:fs';import sharp from '/workspace/ZsRoom-art-batch-e/node_modules/sharp/dist/index.mjs';import crypto from 'node:crypto';
const root='/workspace/ZsRoom-art-batch-g1/art/concepts/batch_g1/v001';fs.mkdirSync(root,{recursive:true});
const records=JSON.parse(fs.readFileSync('/tmp/zs-g1-generation-complete.json'));const selected={small:'A',large:'B',shop:'B',warung:'C'};const widths={small:420,large:600,shop:420,warung:500};const depths={small:320,large:400,shop:500,warung:320};let audit=[];
for(const r of records){let d=`${root}/${r.id}`;fs.mkdirSync(d+'/raw',{recursive:true});fs.mkdirSync(d+'/previews',{recursive:true});fs.copyFileSync(r.source,`${d}/raw/${r.candidate}.png`);fs.writeFileSync(`${d}/raw/${r.candidate}-prompt.md`,r.prompt+'\n\nActual conditioning references:\n'+r.references.join('\n')+'\n');
 const {data,info}=await sharp(r.source).ensureAlpha().raw().toBuffer({resolveWithObject:true});let cleared=0,snapped=0;for(let i=3;i<data.length;i+=4){if(data[i]<=8){if(data[i])cleared++;data[i]=0}else if(data[i]>=250){if(data[i]<255)snapped++;data[i]=255}}
 const norm=await sharp(data,{raw:info}).png().toBuffer();const tr=await sharp(norm).trim({background:{r:0,g:0,b:0,alpha:0},threshold:1}).toBuffer({resolveWithObject:true});
 const m=await sharp(tr.data).extend({top:16,bottom:16,left:16,right:16,background:{r:0,g:0,b:0,alpha:0}}).png().toBuffer();fs.writeFileSync(`${d}/previews/${r.candidate}-native.png`,m);
 if(selected[r.id]===r.candidate){fs.writeFileSync(`${d}/master-native.png`,m);let W=widths[r.id],H=Math.round(tr.info.height*W/tr.info.width);await sharp(tr.data).resize(W,H).extend({top:8,bottom:8,left:8,right:8,background:{r:0,g:0,b:0,alpha:0}}).png().toFile(`${d}/preview-sd.png`);audit.push({id:r.id,selected:r.candidate,rawDimensions:[info.width,info.height],masterDimensions:[tr.info.width+32,tr.info.height+32],contentDimensions:[tr.info.width,tr.info.height],nativeDensity:tr.info.width/W,proposedWidthCm:W,proposedDepthCm:depths[r.id],previewCanvas:[W+16,H+16],previewPivot:[W/2+8,H+7],productionStatus:'BLOCKED: archetype contract and visual approval missing; native density not guaranteed2px/cm',alphaCleared:cleared,alphaSnapped:snapped,sourceSHA256:crypto.createHash('sha256').update(fs.readFileSync(r.source)).digest('hex'),masterSHA256:crypto.createHash('sha256').update(m).digest('hex')});}
}
fs.writeFileSync(root+'/measurements.json',JSON.stringify(audit,null,2));fs.writeFileSync(root+'/generation-record.json',JSON.stringify(records.map(r=>({...r,source:`${r.id}/raw/${r.candidate}.png`})),null,2));console.log(audit);

```
